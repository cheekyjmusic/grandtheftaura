Everything zip -
$repoPath = "C:\Users\jorge\source\repos\cheekyjmusic\grandtheftaura"
$outputZip = Join-Path $repoPath "worldfiles.zip"
$tempDir = Join-Path $env:TEMP "grandtheftaura_temp"

# Clean up any old temp folder or existing zip
if (Test-Path $tempDir) { Remove-Item $tempDir -Recurse -Force }
if (Test-Path $outputZip) { Remove-Item $outputZip -Force }

# Gather index.html, .js, and .css files (skipping node_modules and .git)
$files = @(Get-Item (Join-Path $repoPath "index.html")) + 
         @(Get-ChildItem -Path $repoPath -Recurse -Include *.js, *.css | 
           Where-Object { $_.FullName -notmatch '\\(node_modules|\.git)\\' })

# Copy files to a temporary staging folder while keeping relative paths intact
foreach ($file in $files) {
    $relativePath = $file.FullName.Substring($repoPath.Length).TrimStart('\')
    $destPath = Join-Path $tempDir $relativePath
    $destParent = Split-Path $destPath -Parent
    
    if (-not (Test-Path $destParent)) {
        New-Item -ItemType Directory -Path $destParent -Force | Out-Null
    }
    Copy-Item -Path $file.FullName -Destination $destPath
}

# Create the zip archive and delete the temp folder
Compress-Archive -Path "$tempDir\*" -DestinationPath $outputZip -Force
Remove-Item $tempDir -Recurse -Force
